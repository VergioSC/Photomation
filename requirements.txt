# Photomation dependencies. Install with:
#   .venv/bin/pip install -r requirements.txt
#
# Versions are pinned so the project rebuilds identically on another machine
# (the whole point of checking this file in: .venv/ is gitignored, so this is
# the only record of what the project needs).

opencv-python-headless==5.0.0.93  # image decoding, resizing, Laplacian edge filter.
                          # Headless: we never open a window, and the GUI build drags
                          # in libGL, which costs ~200MB in a container.
numpy==2.5.4              # array maths; OpenCV images are numpy arrays

# Format support beyond JPEG/PNG:
pillow==13.0.0            # cheap image-header reads (size without decoding)
pillow_heif==1.8.1        # HEIC/HEIF, the default iPhone photo format
rawpy==0.28.0             # camera RAW (CR2/CR3/NEF/ARW/DNG/RAF/ORF/RW2/PEF/SRW)

# Web dashboard:
flask==3.1.3              # landing page, dashboard and JSON/SSE API
gunicorn==23.0.0          # the server the container runs. Flask's own is a
                          # development server: single connection at a time, no
                          # request limits, and it says so in its own startup
                          # warning. Not used when running app.py directly.

# Optional: only needed for PHOTOMATION_STORAGE=s3. Left out of the pinned set
# above because boto3 and botocore are ~80MB and a local run never touches
# them. Install with:  pip install boto3
#   boto3
